#!/bin/bash

echo "======================================="
echo "   MySQL Advanced Installation Script  "
echo "======================================="

# System Update
echo "Updating system packages..."
dnf update -y

# Install MySQL Server
echo "Installing MySQL Server..."
dnf install mysql-server -y

# Start MySQL Service
echo "Starting MySQL Service..."
systemctl start mysqld

# Enable MySQL at Boot
echo "Enabling MySQL service at startup..."
systemctl enable mysqld

# Check Service Status
echo "Checking MySQL service status..."
systemctl status mysqld --no-pager

# Extract Temporary Root Password
echo "Getting temporary MySQL root password..."
temp_pass=$(grep 'temporary password' /var/log/mysqld.log | awk '{print $NF}')

echo "Temporary Root Password: $temp_pass"

# Configure Firewall
echo "Opening MySQL port in firewall..."
firewall-cmd --permanent --add-service=mysql
firewall-cmd --reload

# Verify Installation
echo "Verifying MySQL Installation..."
mysql --version

echo "======================================="
echo "MySQL Installation Completed"
echo "======================================="

echo "Next Step:"
echo "Run the command below to secure MySQL"
echo "mysql_secure_installation"
